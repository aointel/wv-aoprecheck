/**
 * Tie 609 number to the dedicated /incomingcall webhook.
 * Sets number Voice URL to https://<baa2>/incomingcall and CLEARS TwiML App.
 * Run: npx tsx server/scripts/set-609-voice-incomingcall.ts

import twilio from 'twilio';
import { TWILIO_ACCOUNT_SID, TWILIO_AUTH_TOKEN } from '../hardcoded-config';

const INBOUND_LAST10 = '6096048379';
const BAA2_BASE = 'https://aoirail-production-baa2.up.railway.app';
const VOICE_URL = `${BAA2_BASE}/incomingcall`;
const STATUS_CALLBACK = `${BAA2_BASE}/api/twilio/call-status`;

async function main() {
  if (!TWILIO_ACCOUNT_SID || !TWILIO_AUTH_TOKEN) { console.error('Missing Twilio credentials'); process.exit(1); }
  const client = twilio(TWILIO_ACCOUNT_SID, TWILIO_AUTH_TOKEN);
  const list = await client.incomingPhoneNumbers.list();
  const target = list.find((n) => String(n.phoneNumber || '').replace(/\D/g, '').slice(-10) === INBOUND_LAST10);
  if (!target) { console.error('No number ending in', INBOUND_LAST10); process.exit(1); }
  await client.incomingPhoneNumbers(target.sid).update({
    voiceUrl: VOICE_URL,
    statusCallback: STATUS_CALLBACK,
    voiceApplicationSid: '',
  });
  console.log('OK:', target.phoneNumber, '-> voiceUrl:', VOICE_URL);
}
main().catch((e) => { console.error(e?.message ?? e); process.exit(1); });

  export function getWorkflowSidForEnqueue(): string {
    const sid = TWILIO_TASKROUTER_WORKFLOW_INBOUND_SID || process.env.TWILIO_TASKROUTER_WORKFLOW_INBOUND_SID;
    if (!sid) throw new Error('TWILIO_TASKROUTER_WORKFLOW_INBOUND_SID not set. Run provisioning script first.');
    return sid;
  }

  export function buildTaskAttributesForEnqueue(input: EnqueueTaskAttributesInput): string {
    const created_at = input.created_at ?? new Date().toISOString();
    const attrs = {
      call_sid: input.call_sid,
      lead_id: input.lead_id ?? '',
      source: input.source,
      market: input.market,
      state: input.state,
      phone_number: input.phone_number,
      lead_name: input.lead_name ?? ([input.first_name, input.last_name].filter(Boolean).join(' ').trim() || 'Unknown'),
      first_name: input.first_name,
      last_name: input.last_name,
      taalk_lead_id: input.taalk_lead_id,
      lead_email: input.lead_email,
      priority_class: input.priority_class ?? 'core',
      transfer_confidence: input.transfer_confidence,
      estimated_value_band: input.estimated_value_band,
      prior_agent_id: input.prior_agent_id,
      workflow_variant: input.workflow_variant,
      fallback_stage: input.fallback_stage ?? 0,
      created_at,
      ...(input.routing_target != null && { routing_target: input.routing_target }),
    };
    return JSON.stringify(attrs);
  }

  export function buildEnqueueTwiML(workflowSid: string, taskAttributesJson: string, waitUrl?: string): string {
    const wait = waitUrl ? ` waitUrl="${escapeXml(waitUrl)}"` : '';
    return `<?xml version="1.0" encoding="UTF-8"?>
<Response>
  <Enqueue workflowSid="${escapeXml(workflowSid)}"${wait}>
    <Task>${escapeXml(taskAttributesJson)}</Task>
  </Enqueue>
</Response>`;
  }
  function escapeXml(s: string): string {
    return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&apos;');
  }

7) FILTERING IN TASKROUTER — ATTRIBUTES MUST MATCH WHAT THE ROUTE EXPECTS
--------------------------------------------------------------------------------
The workflow target expression looks for task.market, task.state, worker.markets, worker.licensed_states.
Whatever we send as task attributes and whatever we put in worker attributes must use these exact keys
and the same value format, or the route won't match.

Task attributes (what we send in Enqueue; route checks task.market, task.state only):
  market: string   — "Unknown" if no lead; otherwise e.g. "Globe Market" (must match worker.markets values)
  state: string    — "XX" if no lead; otherwise 2-letter uppercase e.g. "CO", "TX"

Worker attributes (what we sync; route checks worker.markets, worker.licensed_states):
  markets: string[]             — same canonical strings as task.market (e.g. ["Globe Market"])
  licensed_states: string[]     — 2-letter uppercase (e.g. ["CO", "TX"]) to match task.state
  contact_uri: string

TaskQueue targetWorkers: 1==1. Workflow filter: 1==1. Workflow target expression: 1==1.
No market/state requirement in TaskRouter — any online worker in the queue gets offered the task.
(Task and worker still have market/state for our assignment callback / UI / analytics if we want.)
